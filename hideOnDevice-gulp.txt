function generateSearchIndex(sampleArray) {
    elasticlunr.clearStopWords();
    var instance = elasticlunr(function() {
        this.addField('component');
        this.addField('name');
        this.setRef('uid');
    });
    for (sampleCollection of sampleArray) {
        var component = sampleCollection.name;
        var directory = sampleCollection.directory;
        var puid = sampleCollection.uid;
        var hideOnDevice = sampleCollection.hideOnDevice;
        var ignoreOnBuild = sampleCollection.ignoreOnBuild;
        for (sample of sampleCollection.samples) {
            sample.component = component;
            // Preserve explicit 'dir' set per-sample (e.g. ai-diagram samples inside ai-powered)
            if (!sample.dir) {
                sample.dir = directory;
            }
            sample.parentId = puid;
            sample.hideOnDevice = hideOnDevice ? hideOnDevice : sample.hideOnDevice;
            sample.ignoreOnBuild = ignoreOnBuild ? ignoreOnBuild : sample.ignoreOnBuild;
            instance.addDoc(sample);
        }
    }
    fs.writeFileSync('./src/common/search-index.json', JSON.stringify(instance.toJSON()));
}

function getSamples(data, component, category) {
    var dataList = Object.keys(data);
    for (var i = 0; i < dataList.length; i++) {
        var currentData = data[dataList[i]];
        if (component === currentData.name && category === currentData.category) {
            return currentData;
        }
    }
}

function getControls() {
    var controlList = sampleList;
    var controls = controlList.split(',');
    for (var i = 0; i < controls.length; i++) {
        controls[i] = controls[i].replace(/^\s*/, '').replace(/\s*$/, '');
    }
    return controls;
}

function getSampleOrder(controls) {
    var controlWiseSampleOrder = {};
    var orderKeys = Object.keys(sampleOrder);
    for (var i = 0; i < orderKeys.length; i++) {
        var keys = orderKeys[i];
        //get the controlslist 
        var components = sampleOrder[keys];
        for (j = 0; j < controls.length; j++) {
            // get the index of controls 
            var componentIndex = components.indexOf(controls[j]);
            if (componentIndex != -1) {
                if (!controlWiseSampleOrder[keys]) {
                    controlWiseSampleOrder[keys] = [components[componentIndex]];
                } else {
                    controlWiseSampleOrder[keys].push(components[componentIndex]);
                }
            }
        }
    }
    return controlWiseSampleOrder;
}

gulp.task('combine-samplelist', function() {
    var apiReference = {};
    if (sampleList && sampleList.length) {
        var controls = getControls();
        sampleOrder = getSampleOrder(controls);
    }
    return gulp.src(config.samplejson)
        .pipe(jsoncombine('samplelist.ts', function(data) {
            var result = [];
            var subCategory = [];
            var intId = 0;
            var addUID = function(pid, dt) {
                for (var i = 0; i < dt.length; i++) {
                    dt[i].uid = pid + i;
                    if (dt[i].hasOwnProperty('samples')) {
                        curDirectory = dt[i].directory;
                        subCategory = [];
                        addUID('00' + intId + i, dt[i].samples);
                        intId++;
                    } else {
                        // Preserve explicit 'dir' from sample.json; fall back to parent directory
                        if (!dt[i].dir) {
                            dt[i].dir = curDirectory;
                        }
                        var index = subCategory.indexOf(dt[i].category);
                        if (index !== -1) {
                            dt[i].order = index;
                        } else {
                            subCategory.push(dt[i].category);
                            dt[i].order = subCategory.length - 1;
                        }
                    }
                }
            }
            var orderKeys = Object.keys(sampleOrder);
            for (var i = 0; i < orderKeys.length; i++) {
                var components = sampleOrder[orderKeys[i]];
                let category = orderKeys[i];
                for (var j = 0; j < components.length; j++) {
                    var currentData = getSamples(data, components[j], category);
                    currentData['order'] = i;
                    result.push(currentData);
                }
            }
            addUID("0", result);
            generateSearchIndex(result);
            var configProps = JSON.parse(fs.readFileSync('./config.json', 'utf8'));
            var commonChunkSkip = configProps.cssComponent || [];
            return new Buffer('export let samplesList : any =' + JSON.stringify(result) + ';\n\n' + 'export let apiList:any=' + JSON.stringify(apiReference) + '\n\n export let skipCommonChunk: string[] = ' + JSON.stringify(commonChunkSkip) + ';');
        }))
        .pipe(gulp.dest('./src/common/'));
});
    
gulp.task('generate-routes', function(done) {
    // Dynamically skip all src/ai-*/sample.json files except those that are standalone components
    var jsonFiles = glob.sync('./src/**/sample.json').filter(function(f) {
        var normalized = f.replace(/\\/g, '/');
        // Exclude any src/ai-<anything>/ EXCEPT: ai-powered (unified), ai-assistview, ai-smart-paste, ai-smart-textarea
        return !/\/src\/ai-(?!powered\/|assistview\/|smart-paste\/|smart-textarea\/)[\w-]+\/sample\.json$/.test(normalized);
    });
    var imports = '',
    routs = ["{path: '/', redirect: '/tailwind3/grid/grid-overview'}"];
    jsonFiles.forEach(file => {
        var curJson = require(__dirname + '/' + file);
        curJson.samples.forEach(curSample => {
            // Use sample's own 'dir' if present (for unified entries like ai-powered), else use parent directory
            var sampleDir = curSample.dir || curJson.directory;
            var path = sampleDir + '/' + curSample.url;
            var aData = {}
            // Strip spaces AND hyphens from component name to produce a valid JS identifier
            var compVarName = curJson.name.replace(/[\s\-]/g, '') + curSample.url.split('-').join('');
            imports = imports + 'import ' + compVarName + ' from \'./' + path + '.vue\'; \n';
            aData["eCompName"] = curJson.name;
            aData["eSampleName"] = curSample.name;
            aData["eCategoryName"] = curSample.category;
            routs.push('{ path: \'/:theme/' + path + '.html\', component: ' + compVarName + ', meta: ' + JSON.stringify(aData) + ' }\n');
        });
    });
    fs.writeFileSync("./src/router.config.ts", imports + 'export default [' + routs + '];');
    done();
});

gulp.task('hideOnDevice', function (done) {
    // Replace Vue files with hideOnDevice txt files and create backups
    var filesToReplace = [
        { sourceFile: './hideOnDevice-app.txt', targetFile: './src/App.vue', backupFile: './App.vue.backup' },
        { sourceFile: './hideOnDevice-gulp.txt', targetFile: './gulpfile.js', backupFile: './gulpfile.js.backup' }
    ];
    
    for (var i = 0; i < filesToReplace.length; i++) {
        var file = filesToReplace[i];
        if (fs.existsSync(file.sourceFile)) {
            // Create backup of original file before replacement
            if (fs.existsSync(file.targetFile)) {
                fs.copyFileSync(file.targetFile, file.backupFile);
                console.log(`Backup created: ${file.backupFile}`);
            }
            // Replace with source file content
            var fileContent = fs.readFileSync(file.sourceFile, 'utf8');
            fs.writeFileSync(file.targetFile, fileContent, 'utf8');
            console.log(`Successfully replaced ${file.targetFile} with ${file.sourceFile}`);
        } else {
            console.log(`Warning: Source file ${file.sourceFile} not found in root directory`);
        }
    }
    done();
});
 
gulp.task('revertHideOnDevice', function (done) {
    // Restore Vue files from backups
    var filesToRestore = [
        { targetFile: './src/App.vue', backupFile: './App.vue.backup' },
        { targetFile: './gulpfile.js', backupFile: './gulpfile.js.backup' }
    ];
    
    for (var i = 0; i < filesToRestore.length; i++) {
        var file = filesToRestore[i];
        if (fs.existsSync(file.backupFile)) {
            fs.copyFileSync(file.backupFile, file.targetFile);
            fs.unlinkSync(file.backupFile);
            console.log(`Successfully reverted ${file.targetFile} from backup`);
        } else {
            console.log(`Warning: Backup file ${file.backupFile} not found. No revert performed for ${file.targetFile}`);
        }
    }
    done();
}
